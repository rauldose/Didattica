SUB aprifile ()

   j$ = "Il file " + filenome$ + " non e' in archivio." + CHR$(13) + CHR$(10)

   IF r = 6 THEN t1.text = ""
   EXIT SUB
END IF

IF INSTR(nomefile$, ".TXT") = 0 THEN
   OPEN nomefile$ FOR INPUT AS #1
     IF INSTR(nomefile$, ".L1") > 0 THEN
        LINE INPUT #1, ii$                  'titolo
        titolo.text = LTRIM$(RTRIM$(ii$))
        LINE INPUT #1, ii$                  'password
        ii$ = LEFT$(ii$, 4)
        jh$ = ""
        FOR i% = 1 TO LEN(ii$)
          z1$ = MID$(ii$, i%, 1)
          z1$ = CHR$(ASC(z1$) + 10)
          jh$ = jh$ + z1$
        NEXT
     END IF
   DO WHILE NOT EOF(1)
     LINE INPUT #1, ii$
     decode ii$
     t5$ = t5$ + ii$ + CHR$(13) + CHR$(10)
   LOOP
ELSE   'caso di testo TXT
OPEN nomefile$ FOR INPUT AS #1
    DO WHILE NOT EOF(1)
      LINE INPUT #1, ii$
      t5$ = t5$ + ii$ + CHR$(13) + CHR$(10)
   LOOP
END IF
CLOSE 1
IF jh$ = cod1$ OR cod1$ = codice$ OR jh$ = "0000" THEN
     t1.text = t1.text + t5$
     modificato = falso
      IF mnuimporta.Checked = vero THEN
         mnuimporta.Checked = falso
         t1.selstart = LEN(t1.text)
       END IF
ELSE
MSGBOX "Password non corretta"
END IF
MOUSEPOINTER = 1
EXIT SUB
ErroreApriFile:
MOUSEPOINTER = 1
IF ERR = 72 THEN MSGBOX "Errore !!. Supporto danneggiato": CLOSE #1: EXIT SUB

END SUB

SUB decode (ii$)
    ON LOCAL ERROR GOTO erroredecode
    hh$ = ""
    lk% = LEN(ii$)
    FOR i% = 1 TO lk%
       b% = ASC(MID$(ii$, i%, 1))
        hh$ = hh$ + CHR$(b% + 10)
    NEXT
    ii$ = hh$
EXIT SUB
erroredecode:
RESUME NEXT
END SUB

SUB Salva ()
 ON LOCAL ERROR GOTO SaveFileErr ' Imposta la routine di gestione d'errore.

   j$ = "Non salvo " + filenome$ + " perche' non contiene caratteri." + CHR$(13) + CHR$(10)

   j$ = j$ + "Non cancello il file perche' non e' presente." + CHR$(13) + CHR$(10)
   MSGBOX j$
   EXIT SUB
   END IF
IF DIR$(nomefile$) = filenome$ AND LEN(LTRIM$(RTRIM$(t1.text))) < 5 THEN
  IF INSTR(nomefile$, ".L1") > 0 THEN
       f$ = LEFT$(nomefile$, INSTR(nomefile$, ".L1") - 1)
       n$ = LEFT$(filenome$, INSTR(filenome$, ".L1") - 1)
       m$ = "  ATTENZIONE !!!  " + CHR$(13)

       m$ = m$ + CHR$(13)
       V# = MSGBOX(m$, 4, "")
       IF V# = 7 THEN EXIT SUB ' risposta no
       KILL nomefile$
       testo2.text = "": titolo.text = ""
       IF DIR$(f$ + ".L2") = n$ + ".L2" THEN KILL f$ + ".L2"
       IF DIR$(f$ + ".L3") = n$ + ".L3" THEN KILL f$ + ".L3"
       IF DIR$(f$ + ".L4") = n$ + ".L4" THEN KILL f$ + ".L4"
       IF DIR$(f$ + ".L5") = n$ + ".L5" THEN KILL f$ + ".L5"
       IF DIR$(f$ + ".DAT") = n$ + ".DAT" THEN KILL f$ + ".DAT"
       mnuL1.caption = "  L&1"
       mnuL2.caption = "  L&2"
       mnuL3.caption = "  L&3"
       mnuL4.caption = "  L&4"
       mnuL5.caption = "  L&5"
       mnuDati.caption = "  DATI"
       EXIT SUB
     END IF
    m$ = "  ATTENZIONE:  " + CHR$(13)
    m$ = m$ + " Vuoi eliminare il file " + filenome$ + " ??" + CHR$(13)
    m$ = m$ + CHR$(13)
